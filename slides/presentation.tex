	\begin{frame}[fragile]{Contractibilité des schémas composables}
		\begin{enumerate}
			\small
			\item Montrer que les schémas composables sont propositionnels :
			\begin{theorem}
				Si $\Gamma \vdash A$ est un schéma composable alors il existe un terme $f$ typé $\Gamma \vdash f : A$. \\
				\begin{center}
					\footnotesize
					$\mathtt{PSTm : \left\{n : \mathbb N\right\}} \left\{\Gamma\mathtt{ : Con\ n}\right\} \left\{\mathtt{A : Arr\ n}\right\} \rightarrow \color{red} \mathrm{PS}\ \Gamma\ A \color{black}\rightarrow \color{red}\mathrm{Tm}\ \Gamma\ A\color{black}$	
				\end{center}
			\end{theorem}
			
			\item Montrer que les schémas composables sont contractiles :
			\begin{theorem}
				Si $\Gamma \vdash A$ est un schéma composable alors il existe au plus un terme $f$, à équivalence près, typé $\Gamma \vdash f : A$.
				\begin{center}
					\footnotesize
					$\mathtt{PSEq : \left\{n : \mathbb N\right\}} \left\{\Gamma\mathtt{ : Con\ n}\right\} \left\{\mathtt{A : Arr\ n}\right\} \color{red} \left(ps : \mathrm{PS}\ \Gamma\ A\right) \left(t\ u : \mathrm{Tm}\ \Gamma\ A\right) \color{black} \rightarrow \color{red} t \sim u \color{black}$		
				\end{center}
			\end{theorem}
		\end{enumerate}
		
		
	\end{frame}

				\hypo[]{f \sim g}
				\hypo[]{g \sim h}
				\infer2[($\sim$-trans)]{f \sim h}
			\end{prooftree}
			\vspace{10pt}
		\end{center}
	\end{frame}
	
	\begin{frame}[noframenumbering]{Séquents nécessaires (cas catégories simples)}
		Listes des séquents nécessaires pour montrer la cohérence de $CaTT_1$.
		\begin{enumerate}
			\item $\emptyset \vdash_{ps} x \rightarrow x$
			\item $x \rightarrow y \vdash_{ps} x \rightarrow y$
			\item $f : x \rightarrow y, g : y \rightarrow z \vdash_{ps} x \rightarrow z$
			\item $f : x \rightarrow y \vdash id \cdot f = f$
			\item $f : x \rightarrow y \vdash f \cdot id = f$
			\item $f : x \rightarrow y, g : y \rightarrow z, h : z \rightarrow w \vdash (f \cdot g) \cdot h = f \cdot (g \cdot h)$
		\end{enumerate}
		Arbres de dérivation sur les slides suivantes.
	\end{frame}

				\hypo[]{\Gamma \vdash g : x \rightarrow z}
				\infer2[(pfst)]{\left<f, g\right> \cdot \pi_0 \sim f}

				\hypo[]{\Gamma \vdash g : x \rightarrow z}

			\end{prooftree} \\

	\begin{frame}[noframenumbering]{Séquents essentiels (cas catégories simples)}
		Identité : \hspace{20pt}

		\end{prooftree}\label{id-proof}

		Flèche : \hspace{20pt}
		\begin{prooftree}
			\infer0[(ax)]{\emptyset \vdash_{ps} x \rightarrow x}
			\infer1[(ext)]{x \rightarrow y \vdash_{ps} x \rightarrow y}
		\end{prooftree}\label{fleche-proof}
		
		\vspace{30pt}
		
		Transitivité : \hspace{20pt}
		\begin{prooftree}
			\infer0[(ax)]{\emptyset \vdash_{ps} x \rightarrow x}
			\infer1[(ext)]{x \rightarrow y \vdash_{ps} x \rightarrow y}
			\infer1[(ext)]{x \rightarrow y, y \rightarrow z \vdash_{ps} x \rightarrow z}
		\end{prooftree}\label{trans-proof}
		
	\end{frame}
	
	\begin{frame}[noframenumbering]{Séquents essentiels (cas catégories simples)}
		Axiome composition avec identité à gauche :
		\begin{center}
		\begin{align*}
			\begin{gathered}
				\Pi_{id \cdot f} =
				\begin{prooftree}
					\infer0[]{f : x \rightarrow y \vdash id_x : x \rightarrow x}
					\hypo[]{f : x \rightarrow y \in f : x \rightarrow y}
					\infer1[]{f : x \rightarrow y \vdash f : x \rightarrow y}
					\infer2[]{f : x \rightarrow y \vdash id_x \cdot f : x \rightarrow y}
				\end{prooftree}
			\end{gathered}
		\end{align*}\label{left-id-proof}
		
		\vspace{30pt}
		\begin{prooftree}
			\infer0[]{\emptyset \vdash_{ps} x \rightarrow x}
			\infer1[]{x \rightarrow y \vdash_{ps} x \rightarrow y}
			\hypo[]{\Pi_{id \cdot f}}
			\hypo[]{f : x \rightarrow y \in f : x \rightarrow y}
			\infer1[]{f : x \rightarrow y \vdash f : x \rightarrow y}
			\infer3[]{f : x \rightarrow y \vdash id \cdot f = f}
		\end{prooftree}
		\end{center}
		
	\end{frame}
	
	\begin{frame}[noframenumbering]{Séquents minimales (cas catégories simples)}
		
		Axiome composition avec identité à droite :
		
		\begin{center}
		$\Pi_{f \cdot id} =$

			\infer1[]{f : x \rightarrow y \vdash f : x \rightarrow y}
			\infer0[]{f : x \rightarrow y \vdash id_y : y \rightarrow y}
			\infer2[]{f : x \rightarrow y \vdash f \cdot id_y : x \rightarrow y}
		\end{prooftree}\label{right-id-proof}
		
		\vspace{30pt}
		\begin{prooftree}
			\infer0[]{\emptyset \vdash_{ps} x \rightarrow x}
			\infer1[]{x \rightarrow y \vdash_{ps} x \rightarrow y}
			\hypo[]{\Pi_{f \cdot id}}
			\hypo[]{f : x \rightarrow y \in f : x \rightarrow y}
			\infer1[]{f : x \rightarrow y \vdash f : x \rightarrow y}
			\infer3[]{f : x \rightarrow y \vdash f \cdot id_y = f}
		\end{prooftree} 
		
		\end{center}
	\end{frame}
	
	\begin{frame}[noframenumbering, allowframebreaks, c]{Séquents essentiels : associativité}
		\begin{center}
		\begin{prooftree}
			\hypo[]{\Pi_{ps}}
			\hypo[]{\Pi_{(f \cdot g) \cdot h}}
			\hypo[]{\Pi_{f \cdot (g \cdot h)}}
			\infer3[$(ps^=)$]{\Gamma \vdash (f \cdot g) \cdot h = f \cdot (g \cdot h)}
		\end{prooftree}\label{assoc-proof} \\
		\vspace{20pt}
		
		$\Gamma = f : x \rightarrow y, g : y \rightarrow z, h : z \rightarrow w$ \\
		
		\vspace{15pt}
		
		$\Pi_{ps} =$
		\begin{prooftree}
			\infer0[]{\emptyset \vdash_{ps} x \rightarrow x}
			\infer1[]{x \rightarrow y \vdash_{ps} x \rightarrow y}
			\infer1[]{x \rightarrow y, y \rightarrow z \vdash_{ps} x \rightarrow z}
			\infer1[]{x \rightarrow y, y \rightarrow z, z \rightarrow w \vdash_{ps} x \rightarrow w}
		\end{prooftree}
		\vspace{20pt}
		
		$\Pi_{(f \cdot g) \cdot h} =$
		\begin{prooftree}
			\hypo[]{f : x \rightarrow y \in \Gamma}
			\infer1[]{\Gamma \vdash f : x \rightarrow y}
			\hypo[]{g : y \rightarrow z \in \Gamma}
			\infer1[]{\Gamma \vdash g : y \rightarrow z}
			\infer2[]{\Gamma \vdash f \cdot g : x \rightarrow z}
			\hypo[]{h : z \rightarrow w \in \Gamma}
			\infer1[]{\Gamma \vdash h : z \rightarrow w}
			\infer2[]{\Gamma \vdash (f \cdot g) \cdot h  : x \rightarrow w}
		\end{prooftree}
		\vspace{30pt}
		
		$\Pi_{f \cdot (g \cdot h)} =$
		\begin{prooftree}
			\hypo[]{f : x \rightarrow y \in \Gamma}
			\infer1[]{\Gamma \vdash f : x \rightarrow y}
			\hypo[]{g : y \rightarrow z \in \Gamma}
			\infer1[]{\Gamma \vdash g : y \rightarrow z}
			\hypo[]{h : z \rightarrow w \in \Gamma}
			\infer1[]{\Gamma \vdash h : z \rightarrow w}
			\infer2[]{\Gamma \vdash g \cdot h : y \rightarrow w}
			\infer2[]{\Gamma \vdash f \cdot (g \cdot h)  : x \rightarrow w}
		\end{prooftree}
		\end{center}
	\end{frame}
